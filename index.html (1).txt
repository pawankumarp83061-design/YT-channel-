        window.onerror = function(e) {
          // 检查是否有DOM被渲染（排除script标签）
          const hasDom = function() {
            // 检查body是否有子元素（排除script标签）
            const children = Array.from(document.body.children || []);
            const nonScriptChildren = children.filter(child => child.tagName !== 'SCRIPT');
            if (nonScriptChildren.length > 0) {
              return true;
            }
            // 检查是否有文本内容（排除空白）
            const textContent = document.body.textContent || '';
            const hasText = textContent.trim().length > 0;
            // 检查是否有可见元素（通过计算样式）
            const allElements = document.body.querySelectorAll('*');
            for (let i = 0; i < allElements.length; i++) {
              const el = allElements[i];
              if (el.tagName === 'SCRIPT' || el.tagName === 'STYLE') continue;
              const style = window.getComputedStyle(el);
              if (style.display !== 'none' && style.visibility !== 'hidden' && style.opacity !== '0') {
                return true;
              }
            }
            return hasText;
          };
          
          // 如果页面已经渲染出DOM，则不发送错误信息
          if (!hasDom()) {
            window.parent.postMessage({
              type: 'error',
              error: e
            }, '*')
          }
        }

        function handleOnload() {
          console.log('window.onload 触发', window.innerHeight, Date.now());
          // 检查是否有defer脚本
          const hasDeferScripts = document.querySelectorAll('script[defer]').length > 0;

          console.log('hasDeferScripts33', hasDeferScripts);
          
          if (hasDeferScripts) {
            // 如果有defer脚本，等待它们执行完成
            console.log('hasDeferScripts11', hasDeferScripts);
            setTimeout(function() {
              window.parent.postMessage({
                type: 'onload',
              }, '*');
            }, 1000);
          } else {
            console.log('hasDeferScripts22', hasDeferScripts);
            // 没有defer脚本，直接发送
            window.parent.postMessage({
              type: 'onload',
            }, '*');
          }
        }
        
        // 检查文档是否已经加载完成
        console.log('document.readyState:', document.readyState, 'Date.now():', Date.now());
        
        // 立即检查，如果已经加载完成则立即执行
        if (document.readyState === 'complete') {
          console.log('document already loaded, executing immediately');
          handleOnload();
        } else {
          // 否则等待 window.onload 事件
          console.log('waiting for window.onload, current readyState:', document.readyState);
          // 同时监听 load 事件（更可靠）
          window.addEventListener('load', handleOnload);
          // 也设置 window.onload 作为备用（兼容性）
          if (!window.onload) {

            const oldOnload = window.onload;
            window.onload = function() {
              if (oldOnload) oldOnload();
              handleOnload();
            };

        function selectQuality(quality, element) {
            selectedQuality = quality;
            if (element && element.parentElement) {
                element.parentElement.querySelectorAll('.select-option').forEach(el => el.classList.remove('selected'));
                element.classList.add('selected');
            }
            updateBadges();
            updateTopicDetails();
            showToast(`Quality: ${quality}`, 'success');
        }

        function selectFPS(fps, element) {
            selectedFPS = fps;
            if (element && element.parentElement) {
                element.parentElement.querySelectorAll('.select-option').forEach(el => el.classList.remove('selected'));
                element.classList.add('selected');
            }
            updateBadges();
            updateTopicDetails();
            showToast(`FPS: ${fps}`, 'success');
        }

        function updateBadges() {
            const qualityBadge = document.getElementById('qualityBadge');
            const fpsBadge = document.getElementById('fpsBadge');
            const durationBadge = document.getElementById('durationBadge');
            const durationSelect = document.getElementById('videoDuration');
            if (qualityBadge) qualityBadge.textContent = selectedQuality;
            if (fpsBadge) fpsBadge.textContent = selectedFPS + ' FPS';
            if (durationBadge && durationSelect) durationBadge.textContent = durationSelect.value + 's';
        }

        function downloadHTML() {
            try {
                const htmlContent = document.documentElement.outerHTML;
                const blob = new Blob([htmlContent], { type: 'text/html;charset=utf-8' });
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = 'yt-auto-MBkaeX.html';
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
                URL.revokeObjectURL(url);
                showToast('✅ Download started!', 'success');
            } catch (error) {
                showToast('Try Method 2 (Copy Code)', 'error');
            }
        }

        function copyFullCode() {
            const htmlContent = document.documentElement.outerHTML;
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(htmlContent).then(() => {
                    showToast('✅ Code copied! Paste in Notepad and save as .html', 'success');
                }).catch(() => { fallbackCopy(htmlContent); });
            } else {
                fallbackCopy(htmlContent);
            }
        }

        function fallbackCopy(text) {
            const textarea = document.createElement('textarea');
            textarea.value = text;
            textarea.style.position = 'fixed';
            textarea.style.opacity = '0';
            textarea.style.left = '-9999px';
            document.body.appendChild(textarea);
            textarea.select();
            try {
                document.execCommand('copy');
                showToast('✅ Code copied! Paste in Notepad and save as .html', 'success');
            } catch (err) {
                showToast('Press Ctrl+A then Ctrl+C to copy', 'warning');
            }
            document.body.removeChild(textarea);
        }

        function saveAsFile() {
            showToast('Press Ctrl+S (Windows) or Cmd+S (Mac) to save', 'info');
        }

        function copyShareLink() {
            const linkText = window.location.href;
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(linkText).then(() => {
                    showToast('✅ Link copied!', 'success');
                }).catch(() => { showToast('Link: ' + linkText, 'info'); });
            } else {
                showToast('Link: ' + linkText, 'info');
            }
        }

        function exportWorkflow() {
            const exportData = {
                ...workflowData,
                selectedTopic: selectedTopic,
                settings: {
                    quality: selectedQuality,
                    fps: selectedFPS,
                    duration: document.getElementById('videoDuration')?.value,
                    voice: document.getElementById('voiceType')?.value,
                    animation: document.getElementById('animationStyle')?.value
                },
                exportedAt: new Date().toISOString()
            };
            const dataStr = JSON.stringify(exportData, null, 2);
            const blob = new Blob([dataStr], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = 'workflow-MBkaeX.json';
            a.click();
            URL.revokeObjectURL(url);
            showToast('✅ Workflow exported!', 'success');
        }

        function runWithTopic() {
            if (!selectedTopic) { showToast('Select a topic first!', 'warning'); return; }
            runNow();
        }

        function runNow() {
            if (!selectedTopic) { showToast('Select a topic first!', 'warning'); return; }
            executionCount++;
            const totalVideosEl = document.getElementById('totalVideos');
            if (totalVideosEl) totalVideosEl.textContent = 24 + executionCount;
            showToast(`Starting: ${selectedTopic.title}`, 'info');
            let progress = 0;
            const interval = setInterval(() => {
                progress += 16.67;
                const progressBar = document.getElementById('progressBar');
                const progressText = document.getElementById('progressText');
                if (progressBar) progressBar.style.width = progress + '%';
                if (progressText) progressText.textContent = Math.round(progress) + '%';
                if (progress >= 100) clearInterval(interval);
            }, 800);
            const steps = [
                { delay: 800, message: 'Script generating...' },
                { delay: 1600, message: 'Voice creating...' },
                { delay: 2400, message: 'Video rendering...' },
                { delay: 3200, message: 'Uploading to YouTube...' },
                { delay: 4000, message: '✅ Complete!' }
            ];
            steps.forEach(step => {
                setTimeout(() => showToast(step.message, 'success'), step.delay);
            });
        }

        function openSettings() {
            const modal = document.getElementById('settingsModal');
            if (modal) modal.classList.remove('hidden');
        }

        function closeSettings() {
            const modal = document.getElementById('settingsModal');
            if (modal) modal.classList.add('hidden');
        }

        function saveSettings() {
            const cronInput = document.getElementById('cronInput');
            const nextRunEl = document.getElementById('nextRun');
            if (cronInput && nextRunEl) {
                const cronValue = cronInput.value;
                nextRunEl.textContent = cronValue.split(' ')[1] + ':00 AM';
            }
            closeSettings();
            showToast('Settings saved!', 'success');
        }

        const toggleBtn = document.getElementById('toggleWorkflow');
        if (toggleBtn) {
            toggleBtn.onclick = function() {
                isRunning = !isRunning;
                const indicator = document.getElementById('workflowStatus');
                const text = document.getElementById('statusText');
                if (isRunning) {
                    this.textContent = 'Stop Automation';
                    if (indicator) indicator.className = 'status-dot status-active';
                    if (text) text.textContent = 'Active';
                } else {
                    this.textContent = 'Start Automation';
                    if (indicator) indicator.className = 'status-dot status-inactive';
                    if (text) text.textContent = 'Stopped';
                }
            };
        }

        function showToast(message, type = 'info') {
            const colors = {
                success: 'from-green-500 to-emerald-500',
                error: 'from-red-500 to-rose-500',
                warning: 'from-yellow-500 to-orange-500',
                info: 'from-blue-500 to-indigo-500'
            };
            document.querySelectorAll('.toast').forEach(t => t.remove());
            const toast = document.createElement('div');
            toast.className = `toast fixed bottom-4 right-4 bg-gradient-to-r ${colors[type]} text-white px-6 py-3 rounded-xl shadow-lg z-50 flex items-center gap-3`;
            toast.innerHTML = `
                <svg class="w-5 h-5" viewBox="0 0 24 24" fill="currentColor">
                    <path d="M9 16.17L4.83 12l-1.42 1.41L9 19 21 7l-1.41-1.41z"/>
                </svg>
                <span>${message}</span>
            `;
            document.body.appendChild(toast);
            setTimeout(() => {
                toast.style.opacity = '0';
                toast.style.transition = 'opacity 0.3s';
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }
